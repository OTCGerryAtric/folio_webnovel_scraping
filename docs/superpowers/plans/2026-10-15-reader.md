# Folio Reader Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Read any book inside Folio on any device, resuming at the same paragraph, with your place kept safely on the PC.

**Architecture:** Three separate server modules: `chapters.py` (cleaned chapters as escaped HTML paragraphs with a content version), `anchor.py` (pure position recovery) and `positions.py` (`reading.json` with compare-and-set). Thin routes in `app.py` sit over them. In the browser, four pure modules (`locate`, `sync`, `device`, `settings`) and a network `ChapterSource` sit behind interfaces that Phase 2 can reimplement. One `Reader.svelte` page uses small components.

**Tech Stack:** Python 3.12, FastAPI, pytest; Svelte 5, TypeScript, Tailwind v4, Vitest, @fontsource-variable/literata.

**Spec:** `docs/superpowers/specs/2026-10-15-reader-design.md`

## Global Constraints

- Phase 1 only. No service worker, no IndexedDB, no offline queue.
- **CLI:** behaviour and the existing tests are unchanged.
- **Protections:** every new endpoint is under `/api`, so it inherits the access key, the allowed hosts and the same-origin rule. Nothing is exempted.
- **Escaping:** chapter HTML comes only from the existing renderer, which escapes it. The front end inserts no other HTML.
- **Server-stamped fields:** the server fills `anchor`, `paragraphs`, `version` and `saved_at`. The browser sends `chapter`, `paragraph`, `offset`, `device`, `device_name` and `base_revision`.
- **Paragraphs:** a paragraph is a non-empty block of the chapter body. The body is the file after its first `# ` title line, split on blank lines. This is identical everywhere.
- **Atomic writes:** `reading.json` is written atomically under a per-book lock.
- **Reader settings:** device id and reader settings live in `localStorage`, wrapped in try/catch, with defaults when it is unavailable.
- **Tests:** run with `.venv/Scripts/python.exe -m pytest` and `cd web && npm test`. Visual checks use the scratchpad DevTools driver used for Folio.
- **Commits:** messages end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- **No merge or push** without the user's approval.

```python
"""The reader's view of a book: its cleaned chapters, as escaped HTML paragraphs.

Reads only 02 Cleaned, so a book can be read with gaps and with nothing built.
"""

from __future__ import annotations

import hashlib

from scraper.assemble import block_html
from scraper.book import Book


def chapter_version(text: str) -> str:
    return hashlib.sha1(text.encode("utf-8")).hexdigest()[:12]


def split_chapter(text: str) -> tuple[str | None, list[str]]:
    first, _, rest = text.partition("\n")
    if first.startswith("# "):
        title, body = first[2:].strip() or None, rest
    else:
        title, body = None, text
    return title, [block.strip("\n") for block in body.split("\n\n") if block.strip()]


def _text(book: Book, number: int) -> str | None:
    path = book.cleaned_path(number)
    try:
        return path.read_text(encoding="utf-8")
    except FileNotFoundError:  # not cleaned, or mid re-clean
        return None


def list_chapters(book: Book) -> list[dict]:
    out = []
    for number in book.cleaned_numbers():
        text = _text(book, number)
        if text is None:
            continue
        title, paragraphs = split_chapter(text)
        out.append({"number": number, "title": title, "words": sum(len(p.split()) for p in paragraphs)})
    return out


def load_chapter(book: Book, number: int) -> tuple[str, list[str]] | None:
    text = _text(book, number)
    if text is None:
        return None
    return chapter_version(text), split_chapter(text)[1]


def read_chapter(book: Book, number: int) -> dict | None:
    text = _text(book, number)
    if text is None:
        return None
    title, paragraphs = split_chapter(text)
    numbers = book.cleaned_numbers()
    before = [n for n in numbers if n < number]
    after = [n for n in numbers if n > number]
    return {
        "number": number,
        "title": title,
        "version": chapter_version(text),
        "paragraphs": [h for h in map(block_html, paragraphs) if h],
        "prev": before[-1] if before else None,
        "next": after[0] if after else None,
    }
```

Note: if `block_html` ever returns `""` for a non-blank block (for example a lone `---` rendered differently), paragraph indices would drift. The test `test_split_chapter_takes_the_title_line_off` and the anchor tests depend on `len(paragraphs)` matching. Make `block_html` return `"<hr/>"` for `---`, as `_to_html` does, so there is exactly one HTML entry per markdown paragraph.

```python
"""Finding a saved place in a chapter again, after a re-clean may have changed it. Pure."""

from __future__ import annotations

from scraper.review import normalise

ANCHOR_WORDS = 12
WINDOW = 25
OVERLAP = 0.75


def make_anchor(paragraph: str) -> str:
    return " ".join(normalise(paragraph).split()[:ANCHOR_WORDS])


def _matches(anchor: str, paragraph: str) -> bool:
    opening = make_anchor(paragraph)
    if opening and (opening.startswith(anchor) or anchor.startswith(opening)):
        return True
    want, have = anchor.split(), set(opening.split())
    return bool(want) and sum(w in have for w in want) / len(want) >= OVERLAP


def find_paragraph(anchor: str, paragraphs: list[str], near: int) -> int | None:
    if not anchor:
        return None
    for distance in range(WINDOW + 1):
        for index in {near - distance, near + distance}:
            if 0 <= index < len(paragraphs) and _matches(anchor, paragraphs[index]):
                return index
    return None


def resolve_position(position: dict, numbers: list[int], load) -> dict | None:
    if not numbers:
        return None
    out = dict(position)
    chapter = position["chapter"]
    loaded = load(chapter) if chapter in numbers else None
    if loaded is None:
        later = [n for n in numbers if n > chapter]
        out.update(chapter=later[0] if later else max(numbers), paragraph=0, offset=0.0, recovered="moved")
        return out
    version, paragraphs = loaded
    last = max(len(paragraphs) - 1, 0)
    if version == position.get("version"):
        out.update(paragraph=min(position["paragraph"], last), recovered="exact")
        return out
    found = find_paragraph(position.get("anchor", ""), paragraphs, near=position["paragraph"])
    if found is not None:
        out.update(paragraph=found, recovered="anchor")
        return out
    share = position["paragraph"] / max(position.get("paragraphs") or 1, 1)
    out.update(paragraph=min(int(share * len(paragraphs)), last), offset=0.0, recovered="proportional")
    return out
```

```python
"""Where each book was left: books/<name>/reading.json.

Saves are compare-and-set on a revision number, so a device that has not
seen the latest save cannot overwrite it. Written atomically under a
lock per book.
"""

from __future__ import annotations

import json
import logging
import os
import tempfile
import threading
from datetime import datetime, timezone

from scraper.book import Book

log = logging.getLogger(__name__)
_locks: dict[str, threading.Lock] = {}
_guard = threading.Lock()


class Conflict(Exception):
    def __init__(self, current: dict):
        super().__init__("Another device has saved a newer place.")
        self.current = current


def _lock(book: Book) -> threading.Lock:
    with _guard:
        return _locks.setdefault(str(book.folder.resolve()), threading.Lock())


def read_position(book: Book) -> dict:
    try:
        data = json.loads(book.reading_path.read_text(encoding="utf-8"))
        if isinstance(data, dict) and isinstance(data.get("revision"), int):
            return {"revision": data["revision"], "position": data.get("position")}
    except FileNotFoundError:
        pass
    except (OSError, ValueError):
        log.warning("Ignoring unreadable %s", book.reading_path)
    return {"revision": 0, "position": None}


def save_position(book: Book, base_revision: int, position: dict, now=None) -> dict:
    now = now or (lambda: datetime.now(timezone.utc))
    with _lock(book):
        current = read_position(book)
        if base_revision != current["revision"]:
            raise Conflict(current)
        stamped = {**position, "saved_at": now().strftime("%Y-%m-%dT%H:%M:%SZ")}
        saved = {"revision": current["revision"] + 1, "position": stamped}
        handle, temp = tempfile.mkstemp(dir=book.folder, prefix=".reading-", suffix=".json")
        try:
            with os.fdopen(handle, "w", encoding="utf-8") as out:
                json.dump(saved, out)
            os.replace(temp, book.reading_path)
        except BaseException:
            if os.path.exists(temp):
                os.unlink(temp)
            raise
        return saved
```

```python
class PositionIn(BaseModel):
    base_revision: int
    chapter: int
    paragraph: int
    offset: float
    device: str = Field(max_length=64)
    device_name: str = Field(max_length=80)

    @app.get("/api/books/{name}/chapters")
    def chapters(name: str):
        items = list_chapters(find(name))
        return {"chapters": items, "first": items[0]["number"] if items else None,
                "last": items[-1]["number"] if items else None}

    @app.get("/api/books/{name}/chapters/{number}")
    def chapter(name: str, number: int):
        found = read_chapter(find(name), number)
        if found is None:
            raise HTTPException(404, f"Chapter {number} is not available")
        return found

    def with_resolved(book, stored):
        position = stored["position"]
        resolved = position and resolve_position(position, book.cleaned_numbers(), lambda n: load_chapter(book, n))
        return {**stored, "resolved": resolved}

    @app.get("/api/books/{name}/position")
    def get_position(name: str):
        book = find(name)
        return with_resolved(book, read_position(book))

    @app.put("/api/books/{name}/position")
    def put_position(name: str, body: PositionIn):
        book = find(name)
        loaded = load_chapter(book, body.chapter)
        if loaded is None:
            raise Invalid(f"Chapter {body.chapter} is not available to read")
        version, paragraphs = loaded
        paragraph = max(0, min(body.paragraph, len(paragraphs) - 1))
        position = {
            "chapter": body.chapter, "paragraph": paragraph,
            "offset": max(0.0, min(body.offset, 1.0)),
            "anchor": make_anchor(paragraphs[paragraph]) if paragraphs else "",
            "paragraphs": len(paragraphs), "version": version,
            "device": body.device, "device_name": body.device_name,
        }
        try:
            return with_resolved(book, save_position(book, body.base_revision, position))
        except Conflict as conflict:
            return JSONResponse(with_resolved(book, conflict.current), status_code=409)
```
